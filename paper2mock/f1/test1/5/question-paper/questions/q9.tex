\question{9}{8} It is given that $T_n=\dfrac{n}{n+1}$ and $P_n=T_1\times T_2\times\cdots\times T_n$.

\begin{subparts}
  \item Write $P_5$ as a product of fractions and hence find its value.
  \item Find $P_{20}$ without multiplying all twenty fractions separately.
  \item Given that $P_n=\dfrac{1}{31}$, find the value of $n$.
\end{subparts}

\answerlines{17}
