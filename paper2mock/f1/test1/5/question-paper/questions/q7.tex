\question{7}{8} A rectangular card is $168$ cm long and $126$ cm wide. It is cut into identical squares of the greatest possible side length, with no card left over.

\begin{subparts}
  \item Find the greatest possible side length of each square.
  \item Using the side length in (a),
  \begin{subpartsii}
    \item find the number of squares that can be cut along the length and the width of the paper respectively,
    \item find the total number of squares that can be cut from the paper.
  \end{subpartsii}
\end{subparts}

\answerlines{15}
