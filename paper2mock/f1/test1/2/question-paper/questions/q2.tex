\question{2}{7} Answer the following questions.

\begin{subparts}
  \item In each of the following sequences, find the general term $T_n$.
  \begin{subpartsii}
    \item $2,\ 6,\ 18,\ 54,\ 162,\ \ldots$
    \item $-12,\ -4,\ 4,\ 12,\ 20,\ \ldots$
    \item $125,\ 25,\ 5,\ 1,\ \dfrac{1}{5},\ \ldots$
    \item $-2,\ -\dfrac{1}{3},\ 0,\ \dfrac{1}{7},\ \dfrac{2}{9},\ \ldots$
  \end{subpartsii}

  \item The following table shows the temperature differences between City $P$ and three other cities in winter. ($+1^\circ\text{C}$ means $1^\circ\text{C}$ higher than City $P$.)

  \vspace{1em}

  \par\noindent\hfill
  \begin{tabular}{|c|c|c|c|}
  \hline
  \textbf{City} & Q & R & S \\
  \hline
  \textbf{Temperature difference ($^\circ\text{C}$)} & $+6$ & $-4$ & $-9$ \\
  \hline
  \end{tabular}
  \hfill\mbox{}\par

  \vspace{1em}

  \begin{subpartsii}
    \item If the temperature in City $P$ is $18^\circ\text{C}$, find the temperatures in City $Q$ and City $R$.
    \item If the temperature in City $R$ is $11^\circ\text{C}$, find the temperatures in City $P$ and City $Q$.
  \end{subpartsii}
\end{subparts}
