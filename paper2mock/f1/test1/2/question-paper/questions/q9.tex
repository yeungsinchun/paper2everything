\begin{subparts}
  \item
  \begin{subpartsii}
    \item Write down the first $4$ terms of the sequence.
    \item Find the value of $T_1 + T_2 + \cdots + T_{10}$.
  \end{subpartsii}

  \item
  \begin{subpartsii}
    \item Is the 7th term of the sequence a prime number? Explain your answer.
    \item Is the 10th term of the sequence a prime number? Explain your answer.
  \end{subpartsii}
\end{subparts}
