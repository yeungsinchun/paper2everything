% ------------------------------------------------------------
% Marking scheme subpart.
%
% Each subpart is rendered as its OWN table with just a middle
% vertical strip (no outer border, no horizontal lines anywhere).
% Marks in the right column are LEFT-aligned.
%
% Usage:
%   \begin{mspart}{(a)}
%     \msstep{1 + 1}{1M}    % adds a row inside the table
%     \msstep{= 2}{1A}
%   \end{mspart}
%
% Visual layout (middle vrule only):
%   (a)
%   9/4 - ...        | 1M
%   -19/12...        | 1A
%
% Line spacing strategy:
%   \arraystretch is kept modest so single-line math doesn't
%   balloon up. The extra breathing room between rows is
%   produced by \tabularnewline[14pt] inside \msstep, which
%   adds a fixed 14pt below the row regardless of its content
%   height. This keeps every step visually balanced.
% ------------------------------------------------------------
\renewcommand{\arraystretch}{1.4}

\newenvironment{mspart}[1]{%
  \par\vspace{10pt}\noindent
  \begingroup
  \setlength{\tabcolsep}{4pt}%
  \begin{tabular}{p{0.72\linewidth}|p{0.24\linewidth}}
  \multicolumn{2}{l}{\textbf{#1}}\tabularnewline[6pt]
}{%
  \end{tabular}%
  \endgroup
  \par\vspace{26pt}%
}

% Add a single solution row to the current mspart.
% #1 = math (will be wrapped in $...$)
% #2 = mark, e.g. "1M" or "1A"
%
% \tabularnewline[14pt] leaves a fixed 14pt space below the row,
% regardless of how tall the math content is. This is what
% gives every step a generous, consistent vertical gap.
\newcommand{\msstep}[2]{%
  $#1$ & \textbf{#2}\tabularnewline[14pt]%
}

% Copy the question expression once before working (no marks).
\newcommand{\msexpr}[1]{%
  $#1$ & \tabularnewline[14pt]%
}

% ------------------------------------------------------------
% Standalone, full-width step row (used outside mspart when
% there's no need for borders, e.g. for short text-only steps).
% ------------------------------------------------------------
\newcommand{\msrow}[2]{%
  \par\noindent
  \begin{tabular}{p{0.72\linewidth}|p{0.24\linewidth}}
  $#1$ & \textbf{#2}\\
  \end{tabular}\par\vspace{2pt}%
}
