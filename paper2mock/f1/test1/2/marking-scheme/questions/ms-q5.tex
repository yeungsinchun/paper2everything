\msquestion{5}{4}

\begin{mspart}{(a)}
  \msstep{\text{Digit sum } = 5 + 8 + 3 + 2 = 18, \text{ which is divisible by } 9.}{1A}
\end{mspart}

\begin{mspart}{(b)}
  \msstep{\text{Last three digits are } 07\square; \ 072 \div 8 = 9, \text{ so } \square = 2.}{1A}
\end{mspart}

\begin{mspart}{(c)}
  \msstep{4194 \text{ is not divisible by } 4 \text{ since } 94 \not\equiv 0 \pmod{4}.}{1A}
\end{mspart}

\begin{mspart}{(d)}
  \msstep{\text{Digit sum } = 17 + \square; \ \square = 1, 4 \text{ or } 7.}{1A}
\end{mspart}
