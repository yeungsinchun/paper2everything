\question{9}{8} It is given that $T(n)=n^3$ is the general term of a sequence.

\begin{subparts}
  \item Determine what kind of sequence it is.
  \item The sum of the first $n$ terms of the sequence is given by the following formula:
  \[
  S(n)=1^3+2^3+3^3+4^3+\cdots+n^3=\left[\frac{n(n+1)}{2}\right]^2.
  \]
  \begin{subpartsii}
    \item Find the values of $S(10)$, $S(19)$ and $S(20)$.
    \item Hence, find the values of $8+64+216+512+\cdots+8000$ and $1^3+3^3+5^3+\cdots+19^3$.
  \end{subpartsii}
\end{subparts}

\answerlines{16}
