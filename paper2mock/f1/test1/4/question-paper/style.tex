% Answer space: n ruled lines. Lines span the full text width

  \par\vspace{2mm}
  \foreach \i in {1,...,#1}{%
    \vspace{7mm}\noindent\rule{\AnswerLineWidth}{0.2pt}\par
  }%
}

% ------------------------------------------------------------
% \question{N}{M}: sets the current question number and mark value,
% then writes the inline header "N. " and stores M in the deferred
% marks slot. The stem (if any) follows inline.
% ------------------------------------------------------------
\newlength{\QuestionTextIndent}

\newcommand{\question}[2]{%
  \par\vspace{5mm}%
  \noindent
  \settowidth{\QuestionTextIndent}{\textbf{#1.}\hspace{0.8em}}%
  \hangindent=\QuestionTextIndent\hangafter=1\relax
  \hbox to \QuestionTextIndent{\textbf{#1.}\hspace{0.5em}}%
  \ignorespaces
  \gdef\CurrentQuestionNumber{#1}%
  \gdef\CurrentQuestionMarks{(#2 marks)}%
}

% Stash for the current question's number (used by \subparts) and
% marks (used by \finishquestion / end of \subparts).
\makeatletter
\gdef\CurrentQuestionNumber{}%
\gdef\CurrentQuestionMarks{}%
\makeatother

% ------------------------------------------------------------
% \subparts: enumerate-style block with labels (a), (b), ...
%
% Usage (call \question{N}{M} first to set the number and marks):
%
%   \begin{subparts}                % first \item takes the stem position
%     \item ...
%     \item ...
%   \end{subparts}
%
%   \begin{subparts}{stem}           % explicit stem, then \items follow
%     \item ...
%   \end{subparts}
%
% Layout produced:
%   N. <first item>          (no stem)
%      (a) <item 1>
%      (b) <item 2>
%                       (M marks)
%
%   N. <stem text>           (explicit stem)
%      (a) <item 1>
%      ...
%                       (M marks)
%
% The leading "N. " is printed by \question; \subparts only emits the
% item list and the trailing marks badge.
% ------------------------------------------------------------
\newif\ifSubpartMarksPending
\global\SubpartMarksPendingfalse
\newcounter{enumSub}
\newcounter{enumSubii}
\let\QuestionPaperOriginalItem\item

\makeatletter
\newcommand{\subparts@beginlist}{%
  \setcounter{enumSub}{0}%
  \def\item{%
    \ifnum\value{enumSub}>0\end{minipage}\par\fi
    \refstepcounter{enumSub}%
    \ifnum\value{enumSub}>1\par\vspace{4pt}\fi
    \par\noindent\hspace*{\QuestionTextIndent}%
    \makebox[1.5em][l]{(\alph{enumSub})}%
    \begin{minipage}[t]{\dimexpr\linewidth-\QuestionTextIndent-1.5em\relax}%
  }%
}

\newcommand{\subparts@endlist}{%
  \ifnum\value{enumSub}>0\end{minipage}\par\fi
}

\newenvironment{subparts}{%
  \global\SubpartMarksPendingtrue
  \par\vspace{3pt}%
  \@ifnextchar\bgroup
    {\subparts@withstem}
    {\subparts@nosterm}%
}{%
  \subparts@endlist
  \subparts@finish
}

\newcommand{\subparts@nosterm}{%
  \subparts@beginlist
}

\newcommand{\subparts@withstem}[1]{%
  \par\noindent#1\par\vspace{2pt}%
  \subparts@beginlist
}

\newcommand{\subparts@finish}{%
  \ifSubpartMarksPending
    \global\SubpartMarksPendingfalse
    \par\hfill\textnormal{\CurrentQuestionMarks}\par
  \fi
  \par\vspace{2pt}%
  \gdef\CurrentQuestionMarks{}%
  \ignorespacesafterend
}
\makeatother

\newenvironment{subpartsii}[1][1]{%
  \setcounter{enumSubii}{0}%
  \def\item{%
    \ifnum\value{enumSubii}>0\end{minipage}\par\fi
    \refstepcounter{enumSubii}%
    \ifnum\value{enumSubii}>1\par\vspace{4pt}\fi
    \par\noindent\hspace*{1.5em}%
    \makebox[1.5em][l]{(\roman{enumSubii})}%
    \begin{minipage}[t]{\dimexpr\linewidth-1.5em-1.5em\relax}%
  }%
  \ignorespaces
}{%
  \ifnum\value{enumSubii}>0\end{minipage}\par\fi
}

% ------------------------------------------------------------
% \finishquestion: explicitly drop the deferred marks badge at the
% current position (e.g. mid-question for layout reasons). Also
% clears the auto-drop flag so the badge is not printed again at
% the end of the enclosing subparts block.
% ------------------------------------------------------------
\newcommand{\finishquestion}{%
  \par\ifx\CurrentQuestionMarks\@empty\else
    \global\SubpartMarksPendingfalse
    \hfill\textnormal{\CurrentQuestionMarks}\par
    \gdef\CurrentQuestionMarks{}%
  \fi
}

% Count physical sheets independently of \value{page}, because the
% titlepage environment resets the page counter to 1 (so the cover and the
% first content page both report page=1). Sheet 1 = cover, sheet 2+ = content.
\newcounter{physsheet}
\setcounter{physsheet}{0}
\AtBeginShipout{%
  \stepcounter{physsheet}%
  \ifnum\value{physsheet}>1\relax
    \AtBeginShipoutUpperLeft{%
      \begin{tikzpicture}[remember picture,overlay]
        % Draw the rectangular frame - on ALL content pages (sheet > 1)
        \draw[black,line width=0.4pt]
          ([xshift=17mm,yshift=-22mm]current page.north west) rectangle
          ([xshift=-17mm,yshift=30mm]current page.south east);
        % "Answers written in the margins will not be marked." notice
        % running just inside the left, right and bottom borders.
        \node[black,font=\fontsize{10pt}{10pt}\selectfont,rotate=90,anchor=center]
          at ([xshift=13mm]current page.west)
          {\MarginNotice};
        \node[black,font=\fontsize{10pt}{10pt}\selectfont,rotate=90,anchor=center]
          at ([xshift=-13mm]current page.east)
          {\MarginNotice};
        \node[black,font=\fontsize{10pt}{10pt}\selectfont,anchor=center]
          at ([xshift=-50mm,yshift=25mm]current page.south)
          {\MarginNotice};
      \end{tikzpicture}%
    }%
  \fi
}
