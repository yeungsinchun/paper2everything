\question{9}{8} A sequence satisfies $T_{n+1}=T_n-7$. It is known that $T_8=-19$ and $T_{12}=-47$.
\begin{subparts}
  \item Use $T_8$ to find $T_7$, $T_6$ and $T_5$.
  \item Find $T_1$.
  \item Show that the supplied value of $T_{12}$ is consistent with the recurrence relation.
  \item Find the smallest positive term in the sequence.
\end{subparts}
