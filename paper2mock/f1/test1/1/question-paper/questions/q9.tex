\question{9}{8} It is given that the general term $a_n$ of a sequence is
\[
  a_n = \dfrac{2n}{n+2}.
\]
\begin{subparts}
  \item
  Find the products of
  \begin{subpartsii}
    \item the first $5$ terms,
    \item the first $10$ terms.
  \end{subpartsii}


  \item It is given that the general term $b_n$ of another sequence is
  \[
    b_n = (-1)^n \times \dfrac{n}{n+2}.
  \]
  \noindent Using the result of (a)(ii), find the product of the first $10$ terms.
\end{subparts}

\answerlines{14}
