\msquestion{7}{8}

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|l|c|}
\hline
\textbf{Genres} & \textbf{Number of books} \\
\hline
Science Fiction & $1824$ \\
Mystery         & $1140$ \\
Romance         & $4236$ \\
\hline
\end{tabular}
\end{center}

\vspace{4pt}

\begin{mspart}{(a)}
  \msstep{\text{Mystery} = \dfrac{5}{8} \times 1824 = 1140}{1M}
  \msstep{\text{Romance} = 7200 - 1824 - 1140 = 4236}{1M}
  \msstep{\text{Mystery} = 1140, \ \text{Romance} = 4236}{1A}
\end{mspart}

\begin{mspart}{(b)}
  \msstep{(i) \ \text{HCF}(1824, 1140, 4236) = 12}{1M}
  \msstep{\text{By prime factorisation / successive division.}}{1M}
  \msstep{\text{It is greatest because no larger common divisor of all three numbers exists.}}{1A}
  \msstep{(ii) \ \dfrac{1824}{12} + \dfrac{1140}{12} + \dfrac{4236}{12} = 152 + 95 + 353 = 600}{1M+1A}
\end{mspart}
